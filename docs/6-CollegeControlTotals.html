# --- 2. Fetch & Cache Census Data (The "Extract" Phase) ---
# Downloads and stores ALL counties and ALL age groups for Utah
# This creates a reusable master dataset for future analyses

if not PATHS["census_cache"].exists():
    print(f"Fetching Census data to {PATHS['census_cache']}...")

    decennial_years = [2000, 2010, 2020]
    acs_years = [y for y in range(2009, 2024) if y not in decennial_years]

    data_frames = []

    for year in sorted(decennial_years + acs_years):
        try:
            # Select appropriate dataset and variable codes
            if year in decennial_years:
                dataset = "dec/dhc" if year == 2020 else "dec/sf1"
                code_col = "DHC" if year == 2020 else "SF1"
            else:
                dataset, code_col = "acs/acs5", "ACS"

            year_codes = var_crosswalk[code_col].tolist()
            rename_map = dict(zip(year_codes, var_crosswalk["Label"]))

            # Fetch data from Census API
            df = get_census(
                dataset=dataset,
                variables=["NAME"] + year_codes,
                year=year,
                params={"for": "county:*", "in": "state:49"},
                return_geoid=True,
            )

            # Standardize column names
            df = df.rename(
                columns={c: "NAME" for c in df.columns if c.upper() == "NAME"}
            )
            df = df.rename(columns=rename_map)

            # Keep only relevant columns
            valid_cols = ["NAME"] + var_crosswalk["Label"].tolist()
            df = df[df.columns.intersection(valid_cols)].copy()

            df["Year"], df["Dataset"] = year, dataset
            data_frames.append(df)
            print(f"Fetched {year}")

        except Exception as e:
            print(f"Skipping {year}: {e}")

    # Save combined dataset
    if data_frames:
        PATHS["census_cache"].parent.mkdir(parents=True, exist_ok=True)
        pd.concat(data_frames, ignore_index=True).to_csv(
            PATHS["census_cache"], index=False
        )
        print(f"Census data cached successfully")
else:
    print(f"Using cached Census data from {PATHS['census_cache']}")

    c
    for c in df_ut.columns
    if any(c.startswith(p) for p in target_prefixes)
    and not c.endswith("M")  # Exclude margin of error columns
]

    df_ut[col] = pd.to_numeric(df_ut[col], errors="coerce")

# 3. Melt to long format
df_long = df_ut.melt(
    id_vars=["COUNTY"],
    value_vars=value_cols,
    var_name="Variable",
    value_name="Population",
)

# 4. Parse Year from variable suffix
df_long["Suffix"] = df_long["Variable"].str[5:]
df_long["Year"] = df_long["Suffix"].apply(
    lambda x: int(x) if len(x) == 4 else 2000 + int(x[:2])
)
df_long["Type"] = df_long["Suffix"].apply(
    lambda x: "Decennial" if len(x) == 4 else "ACS"
)

# 5. Filter: Prefer Decennial over ACS for 2010 and 2020
mask_drop = ((df_long["Year"] == 2010) & (df_long["Type"] == "ACS")) | (
    (df_long["Year"] == 2020) & (df_long["Type"] == "ACS")
)
df_clean = df_long[~mask_drop].copy()

# 6. Aggregate by County and Year
df_agg = df_clean.groupby(["COUNTY", "Year"])["Population"].sum().reset_index()

5.1 Regression Analysis with Train/Test Split

Methodology: We use multiple linear regression to predict university enrollment based on county-level college-age populations. To properly validate model performance, we: - Train on data before 2020 (1990-2019) - Test on held-out data (2020-2023) - Include time as a feature to capture secular trends beyond population changes

Show the code 
# --- 1. Prepare Data for Regression ---

# Independent Variables (X): County Populations + Time (Wide Format)
X_hist = df_college_age_hist.set_index("Year").sort_index()
predictors = X_hist.columns.tolist()

# Add time feature (years since start) to capture non-population trends
X_hist["YearsSince2000"] = X_hist.index - 2000

# Update predictor list
predictors_with_time = predictors + ["YearsSince2000"]

# Dependent Variables (y): Enrollment (Wide Format)
# Use only IPEDS data (2000 onwards) for consistency
y_hist_all = df_enroll_ipeds.set_index("Year").sort_index()

# Filter universities that exist in our mapping
universities = [c for c in y_hist_all.columns if c in univ_name_map.keys()]

# Merge to ensure alignment (Inner join on Year)
df_train_master = y_hist_all.join(X_hist, how="inner", rsuffix="_pop")

# --- 2. Split into Train and Test Sets ---
print(
    f"Train/Test Split: Training on years < {TEST_START_YEAR}, Testing on years >= {TEST_START_YEAR}"
)

train_mask = df_train_master.index < TEST_START_YEAR
test_mask = df_train_master.index >= TEST_START_YEAR

df_train = df_train_master[train_mask]
df_test = df_train_master[test_mask]

print(
    f"Training years: {df_train.index.min()} - {df_train.index.max()} ({len(df_train)} years)"
)
print(
    f"Testing years: {df_test.index.min()} - {df_test.index.max()} ({len(df_test)} years)"
)

# --- 3. Train Models & Validate ---
model_metrics_train = []
model_metrics_test = []
model_fit_data = []
residual_data = []
trained_models = {}  # Store models here for the Prediction step

for univ_code in universities:
    # Prepare training data
    train_subset = df_train[[univ_code] + predictors_with_time].dropna()
    test_subset = df_test[[univ_code] + predictors_with_time].dropna()

    if train_subset.empty:
        continue

    X_train = train_subset[predictors_with_time]
    y_train = train_subset[univ_code]

    # Fit Model on training data
    model = LinearRegression()
    model.fit(X_train, y_train)

    # Store trained model
    trained_models[univ_code] = model

    # Predict on Training Data
    y_train_pred = model.predict(X_train)
    train_residuals = y_train - y_train_pred

    # Training Metrics
    r2_train = r2_score(y_train, y_train_pred)
    mape_train = mean_absolute_percentage_error(y_train, y_train_pred)
    rmse_train = np.sqrt(mean_squared_error(y_train, y_train_pred))
    full_name = univ_name_map.get(univ_code, univ_code)

    metrics_train = {
        "University": full_name,
        "Dataset": "Train",
        "R2": float(r2_train),
        "MAPE": float(mape_train),
        "RMSE": float(rmse_train),
        "N": int(len(y_train)),
        "Intercept": float(model.intercept_),
    }

    # Add coefficients
    for pred, coef in zip(predictors_with_time, model.coef_):
        metrics_train[pred] = float(coef)

    model_metrics_train.append(metrics_train)

    # Test Set Validation (if test data exists)
    if not test_subset.empty:
        X_test = test_subset[predictors_with_time]
        y_test = test_subset[univ_code]

        y_test_pred = model.predict(X_test)
        test_residuals = y_test - y_test_pred

        # Test Metrics
        r2_test = r2_score(y_test, y_test_pred)
        mape_test = mean_absolute_percentage_error(y_test, y_test_pred)
        rmse_test = np.sqrt(mean_squared_error(y_test, y_test_pred))

        metrics_test = {
            "University": full_name,
            "Dataset": "Test",
            "R2": float(r2_test),
            "MAPE": float(mape_test),
            "RMSE": float(rmse_test),
            "N": int(len(y_test)),
            "Intercept": float(model.intercept_),
        }

        model_metrics_test.append(metrics_test)

        # Store test predictions for visualization
        for year, actual, pred in zip(test_subset.index, y_test, y_test_pred):
            model_fit_data.append(
                {
                    "Year": int(year),
                    "University": full_name,
                    "Value": float(actual),
                    "Type": "Actual (Test)",
                }
            )
            model_fit_data.append(
                {
                    "Year": int(year),
                    "University": full_name,
                    "Value": float(pred),
                    "Type": "Predicted (Test)",
                }
            )

        # Store test residuals
        for year, resid in zip(test_subset.index, test_residuals):
            residual_data.append(
                {
                    "Year": int(year),
                    "University": full_name,
                    "Residual": float(resid),
                    "Dataset": "Test",
                }
            )

    # Store training predictions for visualization
    for year, actual, pred in zip(train_subset.index, y_train, y_train_pred):
        model_fit_data.append(
            {
                "Year": int(year),
                "University": full_name,
                "Value": float(actual),
                "Type": "Actual (Train)",

                "Year": int(year),
                "University": full_name,
                "Value": float(pred),
                "Type": "Predicted (Train)",
            }

                "Dataset": "Train",

## Regression Analysis with Train/Test Split

**Methodology:** We use multiple linear regression to predict university enrollment based on
county-level college-age populations. To properly validate model performance, we:
- Train on data before 2020 (1990-2019)
- Test on held-out data (2020-2023)
- Include time as a feature to capture secular trends beyond population changes
```{python}
# --- 1. Prepare Data for Regression ---

# Independent Variables (X): County Populations + Time (Wide Format)
X_hist = df_college_age_hist.set_index("Year").sort_index()
predictors = X_hist.columns.tolist()

# Add time feature (years since start) to capture non-population trends
X_hist["YearsSince2000"] = X_hist.index - 2000

# Update predictor list
predictors_with_time = predictors + ["YearsSince2000"]

# Dependent Variables (y): Enrollment (Wide Format)
# Use only IPEDS data (2000 onwards) for consistency
y_hist_all = df_enroll_ipeds.set_index("Year").sort_index()

# Filter universities that exist in our mapping
universities = [c for c in y_hist_all.columns if c in univ_name_map.keys()]

# Merge to ensure alignment (Inner join on Year)
df_train_master = y_hist_all.join(X_hist, how="inner", rsuffix="_pop")

# --- 2. Split into Train and Test Sets ---
print(
    f"Train/Test Split: Training on years < {TEST_START_YEAR}, Testing on years >= {TEST_START_YEAR}"
)

train_mask = df_train_master.index < TEST_START_YEAR
test_mask = df_train_master.index >= TEST_START_YEAR

df_train = df_train_master[train_mask]
df_test = df_train_master[test_mask]

print(
    f"Training years: {df_train.index.min()} - {df_train.index.max()} ({len(df_train)} years)"
)
print(
    f"Testing years: {df_test.index.min()} - {df_test.index.max()} ({len(df_test)} years)"
)

# --- 3. Train Models & Validate ---
model_metrics_train = []
model_metrics_test = []
model_fit_data = []
residual_data = []
trained_models = {}  # Store models here for the Prediction step

for univ_code in universities:
    # Prepare training data
    train_subset = df_train[[univ_code] + predictors_with_time].dropna()
    test_subset = df_test[[univ_code] + predictors_with_time].dropna()

    if train_subset.empty:
        continue

    X_train = train_subset[predictors_with_time]
    y_train = train_subset[univ_code]

    # Fit Model on training data
    model = LinearRegression()
    model.fit(X_train, y_train)

    # Store trained model
    trained_models[univ_code] = model

    # Predict on Training Data
    y_train_pred = model.predict(X_train)
    train_residuals = y_train - y_train_pred

    # Training Metrics
    r2_train = r2_score(y_train, y_train_pred)
    mape_train = mean_absolute_percentage_error(y_train, y_train_pred)
    rmse_train = np.sqrt(mean_squared_error(y_train, y_train_pred))
    full_name = univ_name_map.get(univ_code, univ_code)

    metrics_train = {
        "University": full_name,
        "Dataset": "Train",
        "R2": float(r2_train),
        "MAPE": float(mape_train),
        "RMSE": float(rmse_train),
        "N": int(len(y_train)),
        "Intercept": float(model.intercept_),
    }

    # Add coefficients
    for pred, coef in zip(predictors_with_time, model.coef_):
        metrics_train[pred] = float(coef)

    model_metrics_train.append(metrics_train)

    # Test Set Validation (if test data exists)
    if not test_subset.empty:
        X_test = test_subset[predictors_with_time]
        y_test = test_subset[univ_code]

        y_test_pred = model.predict(X_test)
        test_residuals = y_test - y_test_pred

        # Test Metrics
        r2_test = r2_score(y_test, y_test_pred)
        mape_test = mean_absolute_percentage_error(y_test, y_test_pred)
        rmse_test = np.sqrt(mean_squared_error(y_test, y_test_pred))

        metrics_test = {
            "University": full_name,
            "Dataset": "Test",
            "R2": float(r2_test),
            "MAPE": float(mape_test),
            "RMSE": float(rmse_test),
            "N": int(len(y_test)),
            "Intercept": float(model.intercept_),
        }

        model_metrics_test.append(metrics_test)

        # Store test predictions for visualization
        for year, actual, pred in zip(test_subset.index, y_test, y_test_pred):
            model_fit_data.append(
                {
                    "Year": int(year),
                    "University": full_name,
                    "Value": float(actual),
                    "Type": "Actual (Test)",
                }
            )
            model_fit_data.append(
                {
                    "Year": int(year),
                    "University": full_name,
                    "Value": float(pred),
                    "Type": "Predicted (Test)",
                }
            )

        # Store test residuals
        for year, resid in zip(test_subset.index, test_residuals):
            residual_data.append(
                {
                    "Year": int(year),
                    "University": full_name,
                    "Residual": float(resid),
                    "Dataset": "Test",
                }
            )

    # Store training predictions for visualization
    for year, actual, pred in zip(train_subset.index, y_train, y_train_pred):
        model_fit_data.append(
            {
                "Year": int(year),
                "University": full_name,
                "Value": float(actual),
                "Type": "Actual (Train)",
            }
        )
        model_fit_data.append(
            {
                "Year": int(year),
                "University": full_name,
                "Value": float(pred),
                "Type": "Predicted (Train)",
            }
        )

    # Store training residuals
    for year, resid in zip(train_subset.index, train_residuals):
        residual_data.append(
            {
                "Year": int(year),
                "University": full_name,
                "Residual": float(resid),
                "Dataset": "Train",
            }
        )

# Combine train and test metrics
model_metrics_combined = model_metrics_train + model_metrics_test

# Save diagnostics for reference
pd.DataFrame(model_metrics_combined).to_csv(PATHS["model_diagnostics"], index=False)
print(f"Model diagnostics saved to {PATHS['model_diagnostics']}")

# Convert to proper format for OJS (ensure all values are JSON-serializable)
```
