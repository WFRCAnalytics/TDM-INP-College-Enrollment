# Table 1: Utah School-Age and College-Age Population, 2020-2060

df_gpi_state_forecast = (
    pd.read_excel(
        "data/GPI/SchColAgeFS-Dec2022.xlsx",
        sheet_name="Table 1",
        header=[51, 52],
        nrows=41,
    )
    .dropna(axis=1, how="all")
    .pipe(
        lambda d: d.set_axis(

                "Year",
                "Total_Population",
                "School_Total",

                "School_Growth_Rate",
                "School_Share",
                "College_Total",

            axis=1,
        )
    )

# Source: https://d36oiwf74r1rap.cloudfront.net/wp-content/uploads/2025/11/Gardner-Policy-Institute-State-and-County-Projections-2025-2065-Data.xlsx

# List of target counties for the query
wftdm_counties = [
    "Box Elder County",
    "Weber County",
    "Davis County",
    "Salt Lake County",
    "Utah County",
]

df_gpi_projections = pd.read_excel(
    "data/GPI/Gardner-Policy-Institute-State-and-County-Projections-2025-2065-Data.xlsx",
    sheet_name="Demographic Detail",
    header=3,
    nrows=1230,
).query("Geography in @wftdm_counties")[["Geography", "Year", "Population Ages 18-24"]]

(
    df_gpi_projections.pivot(
        index="Year", columns="Geography", values="Population Ages 18-24"
    )
    .rename_axis(None, axis=1)  # Removes the "Geography" label from the columns
    .reset_index()  # Moves 'Year' from index to the first column
)
