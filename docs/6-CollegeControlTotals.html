# Path to the cached master file
CACHE_PATH = Path("intermediate/Utah_Census_All_Counties_All_Ages_2000_2023.csv")

# --- 2. Fetch & Cache (The "Extract" Phase) ---
if not CACHE_PATH.exists():
    print(f"Fetching Universe to {CACHE_PATH}...")

    decennial_years = [2000, 2010, 2020]
    acs_years = [y for y in range(2009, 2024) if y not in decennial_years]

    data_frames = []

    for year in sorted(decennial_years + acs_years):
        try:
            if year in decennial_years:
                dataset = "dec/dhc" if year == 2020 else "dec/sf1"
                code_col = "DHC" if year == 2020 else "SF1"
            else:
                dataset, code_col = "acs/acs5", "ACS"

            year_codes = var_crosswalk[code_col].tolist()
            rename_map = dict(zip(year_codes, var_crosswalk["Label"]))

            df = get_census(
                dataset=dataset,
                variables=["NAME"] + year_codes,
                year=year,
                params={"for": "county:*", "in": "state:49"},
                return_geoid=True,
            )

            df = df.rename(
                columns={c: "NAME" for c in df.columns if c.upper() == "NAME"}
            )
            df = df.rename(columns=rename_map)

            valid_cols = ["NAME"] + var_crosswalk["Label"].tolist()
            df = df[df.columns.intersection(valid_cols)].copy()

            df["Year"], df["Dataset"] = year, dataset
            data_frames.append(df)
            print(f"Fetched {year}")

        except Exception as e:
            print(f"Skipping {year}: {e}")

    if data_frames:
        CACHE_PATH.parent.mkdir(parents=True, exist_ok=True)
        pd.concat(data_frames, ignore_index=True).to_csv(CACHE_PATH, index=False)

import os
import zipfile
from pathlib import Path
from dotenv import load_dotenv
from ipumspy import IpumsApiClient, AggregateDataExtract, TimeSeriesTable

# Load authentication
load_dotenv()
IPUMS_API_KEY = os.getenv("IPUMS_API_KEY")

if IPUMS_API_KEY:
    client = IpumsApiClient(IPUMS_API_KEY)

    # 1. Define the Time Series Table
    # B57 is the "Persons by Age [18]" table
    ts_table = TimeSeriesTable(
        name="B57",
        geog_levels=["county"],
        years=[
            "1970",
            "1980",
            "1990",
            "2000",
            "2010",
            "2020",
            "2006-2010",
            "2007-2011",
            "2008-2012",
            "2009-2013",

            "2011-2015",
            "2012-2016",
            "2013-2017",
            "2014-2018",
            "2015-2019",
            "2016-2020",
            "2017-2021",
            "2018-2022",
            "2019-2023",
            "2020-2024",
        ],
    )

    # 2. Build the Extract Request
    # Geographic extent "490" represents Utah (FIPS 49 with trailing 0 as per IPUMS spec)
    extract = AggregateDataExtract(
        collection="nhgis",
        description="Utah County College Age Time Series",
        time_series_tables=[ts_table],
        geographic_extents=["490"],
        tst_layout="time_by_column_layout",  # Matches the wide CSV structure
    )

    # 3. Submit and Wait
    print("Submitting extract to IPUMS NHGIS...")
    client.submit_extract(extract)
    print(f"Extract submitted. ID: {extract.extract_id}")

    print("Waiting for extract to complete (this may take a few minutes)...")
    client.wait_for_extract(extract)

    # 4. Download and Extract
    download_dir = Path("data/NHGIS")
    download_dir.mkdir(parents=True, exist_ok=True)

    client.download_extract(extract, download_dir=download_dir)
    zip_path = download_dir / f"nhgis{extract.extract_id:04d}_csv.zip"
    extract_folder = download_dir / f"nhgis{extract.extract_id:04d}_csv"

    print(f"Unzipping to {extract_folder}...")
    with zipfile.ZipFile(zip_path, "r") as zip_ref:
        zip_ref.extractall(extract_folder)

    print("IPUMS NHGIS Fetch Complete!")
else:
    print("Error: IPUMS_API_KEY not found in .env file.")

# Keep rows where State is "Utah"

# 2. Select Target Columns
target_prefixes = ["B57AE", "B57AF", "B57AG", "B57AH"]
value_cols = [
    c
    for c in df_ut.columns
    if any(c.startswith(p) for p in target_prefixes)
    and not c.endswith("M")
]

for col in value_cols:
    df_ut[col] = pd.to_numeric(df_ut[col], errors="coerce")

# 3. Melt
df_long = df_ut.melt(
    id_vars=["COUNTY"],
    value_vars=value_cols,
    var_name="Variable",
    value_name="Population",
)

# 4. Parse Year
df_long["Suffix"] = df_long["Variable"].str[5:]
df_long["Year"] = df_long["Suffix"].apply(
    lambda x: int(x) if len(x) == 4 else 2000 + int(x[:2])
)
df_long["Type"] = df_long["Suffix"].apply(
    lambda x: "Decennial" if len(x) == 4 else "ACS"
)

# 5. Filter (Decennial Priority)
mask_drop = ((df_long["Year"] == 2010) & (df_long["Type"] == "ACS")) | (
    (df_long["Year"] == 2020) & (df_long["Type"] == "ACS")
)
df_clean = df_long[~mask_drop].copy()

# 6. Aggregate
df_agg = df_clean.groupby(["COUNTY", "Year"])["Population"].sum().reset_index()

# 7. Pivot

# Independent Variables (X): County Populations (Wide Format)
X_hist = df_college_age_hist.set_index("Year").sort_index()

# Use .last() to prefer IPEDS data over RTP data for overlapping years (1997-1999)
y_hist_all = df_enroll_combined.groupby("Year").last().sort_index()

# Filter universities that exist in our mapping or are known keys
universities = [
    c for c in y_hist_all.columns
    if c in univ_name_map.keys() or c in ["UOFU", "WSU", "UVU", "SLCC", "BYU"]
]

# Merge to ensure alignment (Inner join on Year)
df_train_master = y_hist_all.join(X_hist, how="inner", rsuffix="_pop")

# --- 2. Train Models & Validate ---
model_metrics = []
model_fit_data = []
trained_models = {}  # Store models here for the Prediction step

for univ_code in universities:
    # Filter valid data for this university (Drop NaNs)
    subset = df_train_master[[univ_code] + predictors].dropna()

    if subset.empty:
        continue

    X = subset[predictors]
    y = subset[univ_code]

    # Fit Model
    model = LinearRegression()
    model.fit(X, y)

    # Store trained model
    trained_models[univ_code] = model

    # Predict on Training Data (Validation)
    y_pred = model.predict(X)

    # Metrics
    r2 = r2_score(y, y_pred)
    mape = mean_absolute_percentage_error(y, y_pred)
    full_name = univ_name_map.get(univ_code, univ_code)

    metrics = {
        "University": full_name,
        "R2": round(r2, 4),
        "MAPE": round(mape, 4),
        "Intercept": round(model.intercept_, 2),
    }
    # Coefficients
    for county, coef in zip(predictors, model.coef_):
        metrics[county] = round(coef, 4)

    model_metrics.append(metrics)

    # Fit Data for Plotting
    for year, actual, pred in zip(subset.index, y, y_pred):
        model_fit_data.append(
            {"Year": year, "University": full_name, "Value": actual, "Type": "Actual"}
        )
        model_fit_data.append(
            {"Year": year, "University": full_name, "Value": pred, "Type": "Predicted"}
        )

# Export to OJS
ojs_define(
    data_model_metrics=pd.DataFrame(model_metrics).to_dict(orient="records"),
    data_model_fit=pd.DataFrame(model_fit_data).to_dict(orient="records"),

# Path to the cached master file
CACHE_PATH = Path("intermediate/Utah_Census_All_Counties_All_Ages_2000_2023.csv")

# --- 2. Fetch & Cache (The "Extract" Phase) ---
if not CACHE_PATH.exists():
    print(f"Fetching Universe to {CACHE_PATH}...")

    decennial_years = [2000, 2010, 2020]
    acs_years = [y for y in range(2009, 2024) if y not in decennial_years]

    data_frames = []

    for year in sorted(decennial_years + acs_years):
        try:
            if year in decennial_years:
                dataset = "dec/dhc" if year == 2020 else "dec/sf1"
                code_col = "DHC" if year == 2020 else "SF1"
            else:
                dataset, code_col = "acs/acs5", "ACS"

            year_codes = var_crosswalk[code_col].tolist()
            rename_map = dict(zip(year_codes, var_crosswalk["Label"]))

            df = get_census(
                dataset=dataset,
                variables=["NAME"] + year_codes,
                year=year,
                params={"for": "county:*", "in": "state:49"},
                return_geoid=True,
            )

            df = df.rename(
                columns={c: "NAME" for c in df.columns if c.upper() == "NAME"}
            )
            df = df.rename(columns=rename_map)

            valid_cols = ["NAME"] + var_crosswalk["Label"].tolist()
            df = df[df.columns.intersection(valid_cols)].copy()

            df["Year"], df["Dataset"] = year, dataset
            data_frames.append(df)
            print(f"Fetched {year}")

        except Exception as e:
            print(f"Skipping {year}: {e}")

    if data_frames:
        CACHE_PATH.parent.mkdir(parents=True, exist_ok=True)
        pd.concat(data_frames, ignore_index=True).to_csv(CACHE_PATH, index=False)

## Regression Analysis & Verification

```{python}
# --- 1. Prepare Data for Regression ---

# Independent Variables (X): County Populations (Wide Format)
X_hist = df_college_age_hist.set_index("Year").sort_index()
predictors = X_hist.columns.tolist()

# Dependent Variables (y): Enrollment (Wide Format)
# Use .last() to prefer IPEDS data over RTP data for overlapping years (1997-1999)
y_hist_all = df_enroll_combined.groupby("Year").last().sort_index()

# Filter universities that exist in our mapping or are known keys
universities = [
    c for c in y_hist_all.columns
    if c in univ_name_map.keys() or c in ["UOFU", "WSU", "UVU", "SLCC", "BYU"]
]

# Merge to ensure alignment (Inner join on Year)
df_train_master = y_hist_all.join(X_hist, how="inner", rsuffix="_pop")

# --- 2. Train Models & Validate ---
model_metrics = []
model_fit_data = []
trained_models = {}  # Store models here for the Prediction step

for univ_code in universities:
    # Filter valid data for this university (Drop NaNs)
    subset = df_train_master[[univ_code] + predictors].dropna()

    if subset.empty:
        continue

    X = subset[predictors]
    y = subset[univ_code]

    # Fit Model
    model = LinearRegression()
    model.fit(X, y)

    # Store trained model
    trained_models[univ_code] = model

    # Predict on Training Data (Validation)
    y_pred = model.predict(X)

    # Metrics
    r2 = r2_score(y, y_pred)
    mape = mean_absolute_percentage_error(y, y_pred)
    full_name = univ_name_map.get(univ_code, univ_code)

    metrics = {
        "University": full_name,
        "R2": round(r2, 4),
        "MAPE": round(mape, 4),
        "Intercept": round(model.intercept_, 2),
    }
    # Coefficients
    for county, coef in zip(predictors, model.coef_):
        metrics[county] = round(coef, 4)

    model_metrics.append(metrics)

    # Fit Data for Plotting
    for year, actual, pred in zip(subset.index, y, y_pred):
        model_fit_data.append(
            {"Year": year, "University": full_name, "Value": actual, "Type": "Actual"}
        )
        model_fit_data.append(
            {"Year": year, "University": full_name, "Value": pred, "Type": "Predicted"}
        )

# Export to OJS
ojs_define(
    data_model_metrics=pd.DataFrame(model_metrics).to_dict(orient="records"),
    data_model_fit=pd.DataFrame(model_fit_data).to_dict(orient="records"),
)
```
